#include <stdio.h>

int main() {
    double a, b;
    char op;

    printf("Unesite izraz (primer: 5 + 3): ");
    scanf("%lf %c %lf", &a, &op, &b);

    switch(op) {
        case '+':
            printf("Rezultat: %.2lf\n", a + b);
            break;
        case '-':
            printf("Rezultat: %.2lf\n", a - b);
            break;
        case '*':
            printf("Rezultat: %.2lf\n", a * b);
            break;
        case '/':
            if (b == 0)
                printf("Greska: deljenje sa nulom!\n");
            else
                printf("Rezultat: %.2lf\n", a / b);
            break;
        default:
            printf("Nepoznata operacija!\n");

    }

    return 0;
}
